<div class="missions-intro">
<p>Le parti pris du Centre de la Photographie Ordinaire : proposer aux visiteurs des expositions dans lesquelles ils se reconnaissent, avec des photos comme celles qu'ils prennent et possèdent. Inviter des artistes contemporains à créer, et des chercheurs à travailler, à partir de la collection, pour nous faire regarder autrement ce qui nous est familier. Tenir un lieu d'expérimentation où culture visuelle populaire et art contemporain sont à égalité, et s'enrichissent mutuellement.</p>
<p>Faire exister ce centre, c'est poser un geste contre l'effacement des photographies ordinaires, contre l'exclusion de certains publics des lieux de culture. Réveiller le regard sur l'ordinaire, c'est restaurer la possibilité de faire entendre les voix du commun des mortels.</p>
</div>

<script>
/* missions — chaque carte se RETOURNE entre deux faces.
   Face avant : titre + bouton .btn-primary « détail » → retourne la carte.
   Face arrière : liste des actions + bouton .btn-primary « retour » → re-retourne.
   L'état vit dans la classe .est-retournee ; aria-expanded suit l'état, et la
   face inactive est rendue inerte (clavier + lecteurs d'écran). */
(function () {
var cartes = Array.prototype.slice.call(document.querySelectorAll('.mission-pave'));

function syncCarte(carte) {
var avant = carte.querySelector('.mission-face-avant');
var arriere = carte.querySelector('.mission-face-arriere');
var bAvant = avant.querySelector('.btn-primary');
var bArriere = arriere.querySelector('.btn-primary');
var retournee = carte.classList.contains('est-retournee');
/* seul le bouton « détail » est un déclencheur dépliable ; le bouton
   « retour » est une simple action, sans aria-expanded */
if (bAvant) bAvant.setAttribute('aria-expanded', String(retournee));
avant.inert = retournee;
arriere.inert = !retournee;
arriere.setAttribute('aria-hidden', String(!retournee));
}

function retourner(carte) {
/* une seule carte retournée à la fois : quand on en ouvre une, les autres
   reviennent à leur face avant (pas d'état « à moitié » dans la page) */
cartes.forEach(function (autre) {
if (autre !== carte && autre.classList.contains('est-retournee')) {
autre.classList.remove('est-retournee');
syncCarte(autre);
}
});
carte.classList.toggle('est-retournee');
syncCarte(carte);
/* le focus suit la face qui vient d'apparaître */
var cible = carte.classList.contains('est-retournee')
? carte.querySelector('.mission-face-arriere .btn-primary')
: carte.querySelector('.mission-face-avant .btn-primary');
if (cible && cible.focus) cible.focus({ preventScroll: true });
}

cartes.forEach(function (carte) {
carte.addEventListener('click', function (e) {
var btn = e.target.closest('.btn-primary');
if (!btn) return;
e.preventDefault();
retourner(carte);
});

/* Échap depuis la face arrière : retour à la face avant */
carte.addEventListener('keydown', function (e) {
if (e.key === 'Escape' && carte.classList.contains('est-retournee')) {
e.preventDefault();
retourner(carte);
}
});

syncCarte(carte);
});
})();
</script>
