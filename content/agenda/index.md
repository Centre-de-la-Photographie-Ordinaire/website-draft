<script>
/* agenda — les vignettes d'une même section sont mises côte à côte dans une
   rangée (.agenda-row). Le HTML source reste une suite d'items (les ateliers
   sont des <details>, les expositions des vignettes-déclencheurs .agenda-item)
   suivie des <dialog> de leurs panneaux : Markdown ne tient pas un <div> qui
   enveloppe des blocs, donc on groupe ici, dans l'ordre. Les panneaux de
   chaque item voyagent avec lui (un modal n'a pas besoin d'être dans la
   rangée, mais il garde ainsi l'ordre du document). */
(function () {
  Array.prototype.forEach.call(document.querySelectorAll('.agenda-list'), function (liste) {
    var rangee = null;
    Array.prototype.slice.call(liste.children).forEach(function (el) {
      if (el.tagName === 'H2' || el.tagName === 'H3') { rangee = null; return; }
      if (!el.matches('.agenda-item')) return;
      if (!rangee) {
        rangee = document.createElement('div');
        rangee.className = 'agenda-row';
        liste.insertBefore(rangee, el);
      }
      rangee.appendChild(el);
      var next = rangee.nextElementSibling;
      while (next && next.tagName === 'DIALOG') { rangee.appendChild(next); next = rangee.nextElementSibling; }
    });
  });
})();
</script>
